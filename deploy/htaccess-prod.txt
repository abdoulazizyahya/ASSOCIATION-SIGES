# =====================================================================
#  Modèle de .htaccess pour la PRODUCTION (Camoo / cPanel, Apache).
#  À copier tel quel en  .htaccess  À LA RACINE du sous-domaine
#  (promeducam.beero.cm). Ne PAS l'utiliser en LAN / WAMP.
# =====================================================================

# NE PAS ajouter « DirectoryIndex » ici : plusieurs mutualisés (Camoo…)
# l'interdisent en .htaccess (AllowOverride restreint) → « Internal Server
# Error / DirectoryIndex not allowed here ». La vitrine publique est servie
# par index.php lui-même (cf. son en-tête) ; « Se connecter » ouvre
# index.php?login → login.php.

# ── Forcer HTTPS ────────────────────────────────────────────────────
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteCond %{HTTPS} off
    RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
</IfModule>

# ── Anciennes adresses → nouvelle arborescence (réorganisation du 12/10/2026) ──
#  Favoris et liens anciens : redirection permanente vers le nouvel
#  emplacement. Le motif est relatif au dossier de ce .htaccess ; la
#  condition calcule la base de l'application (/SIGES/ en LAN, / en
#  production) dans %2. Les QR imprimés (verif_*.php) n'ont pas bougé.
<IfModule mod_rewrite.c>
    RewriteEngine On
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^secondaire/pages/(notes|bulletins)/(.*)$ %2secondaire/general/pages/$1/$2 [R=308,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^secondaire/pages/(.*)$ %2secondaire/commun/pages/$1 [R=308,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^secondaire/pdf/(.*)$ %2secondaire/commun/pdf/$1 [R=308,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^(pages|pdf)/(.*)$ %2primaire/$1/$2 [R=308,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^(notes_apc|notes_apc_arabe)\.php$ %2primaire/$1.php [R=308,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^ajax/(.*)$ %2noyau/ajax/$1 [R=307,L]
    # Racine réduite à index.php (12/10/2026). QR IMPRIMÉS : verif_X.php -> verif/X.php
    # (ne jamais retirer cette règle — les documents papier en circulation en dépendent).
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^verif_(attestation|bulletin|carte|honneur|paiement|recu|scolarite)\.php$ %2verif/$1.php [R=308,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^verif_bulletin_hors_ligne\.html$ %2verif/bulletin_hors_ligne.html [R=308,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^(login|logout|mot_de_passe_oublie|configurer_securite|questions_securite_reporter|profil)\.php$ %2compte/$1.php [R=307,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^dashboard\.php$ %2noyau/dashboard.php [R=307,L]
    RewriteCond $0#%{REQUEST_URI} ^([^#]*)#(.*)\1$
    RewriteRule ^install\.php$ %2installation/install.php [R=307,L]
</IfModule>

# ── PHP : pas d'affichage d'erreurs, journalisation active ──────────
#  (si le serveur utilise PHP-FPM, régler plutôt via cPanel → MultiPHP INI Editor)
<IfModule mod_php.c>
    php_flag display_errors Off
    php_flag log_errors On
</IfModule>
<IfModule mod_php7.c>
    php_flag display_errors Off
    php_flag log_errors On
</IfModule>

# ── Fichiers sensibles : accès direct interdit ─────────────────────
<FilesMatch "(^\.|\.(sql|md|log|example|dist|lock|json|ya?ml|sh|bat)$|^config(\.local)?\.php$)">
    Require all denied
</FilesMatch>

# ── Dossiers internes : accès web interdit ─────────────────────────
#  bd/ contient les runners (installer, migrations, sauvegardes, pool).
#  Les débloquer ponctuellement (par IP) le temps d'une opération d'admin,
#  puis re-commenter.
RedirectMatch 403 ^/(bd|\.git|deploy)(/|$)

# ── Cache statique léger ──────────────────────────────────────────
<IfModule mod_expires.c>
    ExpiresActive On
    ExpiresByType image/png            "access plus 30 days"
    ExpiresByType image/jpeg           "access plus 30 days"
    ExpiresByType image/svg+xml        "access plus 30 days"
    ExpiresByType text/css             "access plus 7 days"
    ExpiresByType application/javascript "access plus 7 days"
    ExpiresByType font/woff2           "access plus 30 days"
</IfModule>
